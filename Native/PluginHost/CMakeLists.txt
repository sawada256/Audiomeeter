cmake_minimum_required(VERSION 3.22)
project(VoicemeeterPluginHost VERSION 1.0.0 LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0")
set(CMAKE_OSX_ARCHITECTURES "arm64")
set(CMAKE_CXX_STANDARD 17)
add_subdirectory("${JUCE_SOURCE}" JUCE EXCLUDE_FROM_ALL)
add_library(VMPluginHost SHARED PluginHost.cpp MacEditorSupport.mm)
target_compile_definitions(VMPluginHost PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_AU=0 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=0)
target_link_libraries(VMPluginHost PRIVATE juce::juce_audio_utils)
option(VM_BUILD_TEST_PLUGIN "Build deterministic integration test VST3" OFF)
if(VM_BUILD_TEST_PLUGIN)
    juce_add_plugin(VMTestGain COMPANY_NAME "Voicemeeter Tests" BUNDLE_ID "local.voicemeeter.testgain" PLUGIN_MANUFACTURER_CODE Vmct PLUGIN_CODE Vmtg FORMATS VST3 PRODUCT_NAME "Voicemeeter Test Gain" COPY_PLUGIN_AFTER_BUILD FALSE)
    target_sources(VMTestGain PRIVATE ../../Tests/TestGainPlugin.cpp)
    target_compile_definitions(VMTestGain PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
    target_link_libraries(VMTestGain PRIVATE juce::juce_audio_utils)
    juce_add_plugin(VMTestSide COMPANY_NAME "Voicemeeter Tests" BUNDLE_ID "local.voicemeeter.testside" PLUGIN_MANUFACTURER_CODE Vmct PLUGIN_CODE Vmts FORMATS VST3 PRODUCT_NAME "Voicemeeter Test Sidechain" COPY_PLUGIN_AFTER_BUILD FALSE)
    target_sources(VMTestSide PRIVATE ../../Tests/TestSidechainPlugin.cpp)
    target_compile_definitions(VMTestSide PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
    target_link_libraries(VMTestSide PRIVATE juce::juce_audio_utils)
endif()
set_target_properties(VMPluginHost PROPERTIES INSTALL_NAME_DIR "@rpath")
add_executable(VMPluginWorker PluginWorker.cpp PluginHost.cpp MacEditorSupport.mm)
target_compile_definitions(VMPluginWorker PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_AU=0 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=0)
target_link_libraries(VMPluginWorker PRIVATE juce::juce_audio_utils)
set_target_properties(VMPluginWorker PROPERTIES BUILD_RPATH "@loader_path/../Frameworks;@loader_path/../../../../Frameworks;@loader_path")
